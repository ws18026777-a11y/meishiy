function saveToGitHub(data) { var token = document.getElementById('tokenInput').value.trim(); if (!token) { showToast('请先输入 GitHub Token', true); return Promise.reject('No token'); } // 修复中文乱码：用 TextEncoder 确保 UTF-8 var content = JSON.stringify(data, null, 2); var encoder = new TextEncoder(); var dataBytes = encoder.encode(content); var encoded = btoa(String.fromCharCode.apply(null, dataBytes)); return fetch('https://api.github.com/repos/' + REPO_OWNER + '/' + REPO_NAME + '/contents/' + FILE_PATH, { method: 'PUT', headers: { 'Authorization': 'token ' + token, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: '更新卡密数据 - ' + new Date().toLocaleString(), content: encoded, sha: currentSha, branch: BRANCH }) }) .then(function(res) { if (!res.ok) { if (res.status === 401) throw new Error('Token 无效'); throw new Error('HTTP ' + res.status); } return res.json(); }) .then(function(data) { currentSha = data.content.sha; showToast('✅ 保存成功！'); return data; }); }
